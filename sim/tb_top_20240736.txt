
`timescale 1ns/1ps

module tb_top_20240736;

    reg a, b, c, d;
    reg [3:0] A, B;
    reg clk, rst, en;

    wire u, v;
    wire [3:0] Y, Q;
    wire flag_comb, flag_q;

    integer i, j, k;
    integer tests, errors;

    reg eu, ev, ef;
    reg [3:0] ey;
    reg [4:0] temp_sum;
    reg [3:0] temp_xor;
    reg [3:0] ctrl;

    top_20240736 dut (
        .a(a), .b(b), .c(c), .d(d),
        .A(A), .B(B),
        .clk(clk), .rst(rst), .en(en),
        .u(u), .v(v),
        .Y(Y), .flag_comb(flag_comb),
        .Q(Q), .flag_q(flag_q)
    );

    // Modelo independiente de las funciones de control
    task reference_control;
        input [3:0] x;
        output ru, rv;
        begin
            case (x)
                4'd2, 4'd3, 4'd5, 4'd7,
                4'd9, 4'd10, 4'd11, 4'd13,
                4'd15: ru = 1'b1;
                default: ru = 1'b0;
            endcase

            case (x)
                4'd4, 4'd5, 4'd8, 4'd10,
                4'd11, 4'd12, 4'd13, 4'd14,
                4'd15: rv = 1'b1;
                default: rv = 1'b0;
            endcase
        end
    endtask

    // Modelo de referencia de la ruta de datos
    task reference_datapath;
        input [3:0] x, y;
        input ru, rv;
        output [3:0] result;
        output rf;
        reg [4:0] s;
        reg [3:0] xr;
        begin
            s = {1'b0, x} + {1'b0, y};
            xr = x ^ y;

            case ({rv, ru})
                2'b00: begin
                    result = x - y;
                    rf = (x < y);
                end
                2'b01: begin
                    result = (x >= y) ? x : y;
                    rf = (x == y);
                end
                2'b10: begin
                    result = xr;
                    rf = ^xr;
                end
                2'b11: begin
                    result = s[3:0];
                    rf = s[4];
                end
            endcase
        end
    endtask

    // Reloj de simulacion
    initial clk = 1'b0;
    always #5 clk = ~clk;

    initial begin
        $dumpfile("reto12.vcd");
        $dumpvars(0, tb_top_20240736);

        tests = 0;
        errors = 0;
        rst = 1'b1;
        en = 1'b0;
        a = 0; b = 0; c = 0; d = 0;
        A = 0; B = 0;

        #2;
        if (Q !== 4'b0000 || flag_q !== 1'b0)
            errors = errors + 1;

        #10;
        rst = 1'b0;

        // 16 controles x 16 valores de A x 16 valores de B
        for (i = 0; i < 16; i = i + 1) begin
            ctrl = i;
            {a,b,c,d} = ctrl;

            for (j = 0; j < 16; j = j + 1) begin
                A = j;

                for (k = 0; k < 16; k = k + 1) begin
                    B = k;
                    #1;

                    reference_control(ctrl, eu, ev);
                    reference_datapath(A, B, eu, ev, ey, ef);

                    tests = tests + 1;

                    if ({u,v,Y,flag_comb} !==
                        {eu,ev,ey,ef}) begin
                        errors = errors + 1;
                        $display(
                            "ERROR ctrl=%d A=%d B=%d",
                            i, j, k
                        );
                        $display(
                            "Esperado u=%b v=%b Y=%h flag=%b",
                            eu, ev, ey, ef
                        );
                        $display(
                            "Obtenido u=%b v=%b Y=%h flag=%b",
                            u, v, Y, flag_comb
                        );
                    end
                end
            end
        end

        $display("--------------------------------");
        $display("Vectores comprobados: %0d", tests);
        $display("Errores: %0d", errors);
        $display("--------------------------------");

        if (tests != 4096 || errors != 0)
            $display("RESULTADO: FALLA");
        else
            $display("RESULTADO: PRUEBA COMBINACIONAL CORRECTA");

        // Prueba temporal de captura y retencion
        @(negedge clk);
        rst = 1'b1;
        en = 1'b1;
        #1;
        if (Q !== 0 || flag_q !== 0)
            errors = errors + 1;

        @(negedge clk);
        rst = 1'b0;
        A = 4'd3;
        B = 4'd5;
        {a,b,c,d} = 4'd0;  // RESTA
        en = 1'b1;

        @(posedge clk);
        #1;
        if (Q !== 4'd14 || flag_q !== 1'b1)
            errors = errors + 1;

        @(negedge clk);
        en = 1'b0;
        A = 4'd1;
        B = 4'd1;

        @(posedge clk);
        #1;
        if (Q !== 4'd14 || flag_q !== 1'b1)
            errors = errors + 1;

        $display("Errores finales, incluyendo pruebas temporales: %0d",
                 errors);

        $finish;
    end

endmodule
